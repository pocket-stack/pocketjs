cmake_minimum_required(VERSION 3.16)
project(pocket_reactivity C)
set(CMAKE_C_STANDARD 11)
option(REACTIVITY_BUILD_RUNNER "Build the standalone QuickJS runner" ON)
add_library(reactivity STATIC core/reactivity.c)
target_include_directories(reactivity PUBLIC core)
target_compile_options(reactivity PRIVATE -Wall -Wextra -Werror)
if(REACTIVITY_BUILD_RUNNER)
  set(QUICKJS_DIR "" CACHE PATH "Extracted QuickJS 2025-09-13 source directory")
  if(NOT EXISTS "${QUICKJS_DIR}/quickjs.c")
    message(FATAL_ERROR "Set QUICKJS_DIR to QuickJS 2025-09-13 sources (release archive -2)")
  endif()
  add_library(experiment_quickjs STATIC
    ${QUICKJS_DIR}/quickjs.c ${QUICKJS_DIR}/dtoa.c ${QUICKJS_DIR}/libregexp.c
    ${QUICKJS_DIR}/libunicode.c ${QUICKJS_DIR}/cutils.c)
  target_include_directories(experiment_quickjs SYSTEM PUBLIC ${QUICKJS_DIR})
  target_compile_definitions(experiment_quickjs PRIVATE _GNU_SOURCE CONFIG_VERSION="2025-09-13")
  target_link_libraries(experiment_quickjs PUBLIC m pthread ${CMAKE_DL_LIBS})
  add_executable(reactivity-runner bindings/quickjs/runner.c)
  target_link_libraries(reactivity-runner PRIVATE reactivity experiment_quickjs)
  target_compile_options(reactivity-runner PRIVATE -Wall -Wextra)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(reactivity-core-test tests/core.c)
  target_link_libraries(reactivity-core-test PRIVATE reactivity)
  target_compile_options(reactivity-core-test PRIVATE -UNDEBUG)
  add_test(NAME reactivity-core COMMAND reactivity-core-test)
endif()
